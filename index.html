</head>

<body class="bg-slate-950 text-white min-h-screen flex flex-col items-center justify-center p-4">

    <div id="loginScreen" class="w-full max-w-md bg-slate-900 p-6 rounded-2xl shadow border border-slate-800 space-y-4">

        <h1 class="text-2xl font-bold text-center text-cyan-400">Klasse mot klasse</h1>

        <div>

            <label class="block text-sm mb-1">Ditt navn:</label>

            <input type="text" id="studentName" placeholder="Skriv navnet ditt..." class="w-full p-3 rounded bg-slate-950 border border-slate-700 text-white">

        </div>

        <div>

            <label class="block text-sm mb-1">Velg din klasse:</label>

            <select id="classSelect" class="w-full p-3 rounded bg-slate-950 border border-slate-700 text-white">

                <option value="">Laster klasser...</option>

            </select>

        </div>

        <button onclick="registerStudent()" class="w-full bg-cyan-600 hover:bg-cyan-500 py-3 rounded-xl font-bold text-lg">Bli med i spillet</button>

    </div>


    <div id="gameScreen" class="hidden w-full max-w-md bg-slate-900 p-6 rounded-2xl shadow border border-slate-800 text-center space-y-4">

        <h2 class="text-xl font-semibold text-yellow-400" id="statusText">Venter på neste spørsmål...</h2>

    </div>


    <script type="module">

        import { supabase } from './supabaseClient.js';


        async function init() {

            // Hent aktiv konkurranse (vi antar den første aktive for enkelhets skyld, eller en fastsatt ID)

            const { data: comps } = await supabase.from('competitions').select('*').limit(1);

            if (!comps || comps.length === 0) return;

            const comp = comps[0];

            window.activeCompId = comp.id;


            if (comp.active_class_ids && comp.active_class_ids.length > 0) {

                const { data: classes } = await supabase.from('classes').select('*').in('id', comp.active_class_ids);

                const select = document.getElementById('classSelect');

                select.innerHTML = '<option value="">-- Velg klasse --</option>' + (classes || []).map(c => `

                    <option value="${c.name}">${c.name}</option>

                `).join('');

            }

        }


        window.registerStudent = async function() {

            const name = document.getElementById('studentName').value.trim();

            const className = document.getElementById('classSelect').value;

            if (!name || !className) {

                alert('Vennligst skriv navn og velg klasse.');

                return;

            }


            const { data, error } = await supabase.from('participants').insert([{

                name,

                class_name: className,

                competition_id: window.activeCompId,

                score: 0

            }]).select();


            if (!error && data) {

                localStorage.setItem('participant_id', data[0].id);

                document.getElementById('loginScreen').classList.add('hidden');

                document.getElementById('gameScreen').classList.remove('hidden');

            }

        }


        init();

    </script>

</body>
