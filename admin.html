</head>

<body class="bg-slate-900 text-white min-h-screen p-6">

    <div class="max-w-4xl mx-auto space-y-8">

        <h1 class="text-3xl font-bold text-yellow-400">Arrangørpanel - Klasse mot klasse</h1>


        <!-- Konkurranse- og klasseadministrasjon -->

        <div class="bg-slate-800 p-6 rounded-xl shadow border border-slate-700 space-y-4">

            <h2 class="text-xl font-semibold text-cyan-400">1. Administrer Konkurranse & Klasser</h2>

            <div class="grid grid-cols-1 md:grid-cols-2 gap-4">

                <div>

                    <label class="block text-sm mb-1">Opprett ny konkurranse:</label>

                    <input type="text" id="compTitle" placeholder="F.eks. Høstquiz 8. trinn" class="w-full p-2 rounded bg-slate-900 border border-slate-700 text-white mb-2">

                    <button onclick="createCompetition()" class="bg-cyan-600 hover:bg-cyan-500 px-4 py-2 rounded font-bold w-full">Opprett Konkurranse</button>

                </div>

                <div>

                    <label class="block text-sm mb-1">Legg til ny klasse (f.eks. 8A):</label>

                    <div class="flex gap-2">

                        <input type="text" id="classNameInput" placeholder="Klassenavn" class="w-full p-2 rounded bg-slate-900 border border-slate-700 text-white">

                        <button onclick="addClass()" class="bg-emerald-600 hover:bg-emerald-500 px-4 py-2 rounded font-bold">Legg til</button>

                    </div>

                </div>

            </div>


            <div>

                <label class="block text-sm mb-1 font-semibold mt-4">Velg aktiv konkurranse:</label>

                <select id="competitionSelect" onchange="loadActiveCompetition()" class="w-full p-2 rounded bg-slate-900 border border-slate-700 text-white">

                    <option value="">-- Velg konkurranse --</option>

                </select>

            </div>


            <div id="classSelectionContainer" class="hidden mt-4 pt-4 border-t border-slate-700">

                <h3 class="text-md font-semibold mb-2">Velg hvilke klasser som deltar i denne konkurransen:</h3>

                <div id="classListCheckboxes" class="flex flex-wrap gap-4 mb-4"></div>

                <button onclick="saveActiveCompetitionClasses()" class="bg-yellow-600 hover:bg-yellow-500 px-4 py-2 rounded font-bold text-black">Lagre deltakende klasser</button>

            </div>

        </div>


        <!-- Spillkontroller (Skjules til konkurranse er valgt) -->

        <div id="gameControls" class="bg-slate-800 p-6 rounded-xl shadow border border-slate-700 space-y-4 hidden">

            <h2 class="text-xl font-semibold text-yellow-400">2. Styr Spillet</h2>

            <div class="flex gap-4">

                <button onclick="nextQuestion()" class="bg-green-600 hover:bg-green-500 px-6 py-3 rounded font-bold text-lg">Neste Spørsmål</button>

                <button onclick="lockAnswers()" class="bg-red-600 hover:bg-red-500 px-6 py-3 rounded font-bold text-lg">Lås Svar</button>

            </div>

        </div>

    </div>


    <script type="module">

        import { supabase } from './supabaseClient.js';


        window.loadCompsAndClasses = async function() {

            // Hent klasser

            const { data: classes } = await supabase.from('classes').select('*');

            const box = document.getElementById('classListCheckboxes');

            box.innerHTML = (classes || []).map(c => `

                <label class="flex items-center gap-2 cursor-pointer">

                    <input type="checkbox" value="${c.id}" class="comp-class-cb w-5 h-5"> ${c.name}

                </label>

            `).join('');


            // Hent konkurranser

            const { data: comps } = await supabase.from('competitions').select('*');

            const select = document.getElementById('competitionSelect');

            select.innerHTML = '<option value="">-- Velg konkurranse --</option>' + (comps || []).map(co => `

                <option value="${co.id}">${co.title}</option>

            `).join('');

        }


        window.addClass = async function() {

            const name = document.getElementById('classNameInput').value.trim();

            if (!name) return;

            await supabase.from('classes').insert([{ name }]);

            document.getElementById('classNameInput').value = '';

            loadCompsAndClasses();

        }


        window.createCompetition = async function() {

            const title = document.getElementById('compTitle').value.trim();

            if (!title) return;

            await supabase.from('competitions').insert([{ title, status: 'setup' }]);

            document.getElementById('compTitle').value = '';

            loadCompsAndClasses();

        }


        window.loadActiveCompetition = async function() {

            const id = document.getElementById('competitionSelect').value;

            const container = document.getElementById('classSelectionContainer');

            const controls = document.getElementById('gameControls');

            if (!id) {

                container.classList.add('hidden');

                controls.classList.add('hidden');

                return;

            }

            container.classList.remove('hidden');

            controls.classList.remove('hidden');


            const { data } = await supabase.from('competitions').select('*').eq('id', id).single();

            if (data && data.active_class_ids) {

                document.querySelectorAll('.comp-class-cb').forEach(cb => {

                    cb.checked = data.active_class_ids.includes(cb.value);

                });

            }

        }


        window.saveActiveCompetitionClasses = async function() {

            const id = document.getElementById('competitionSelect').value;

            const selected = Array.from(document.querySelectorAll('.comp-class-cb:checked')).map(cb => cb.value);

            await supabase.from('competitions').update({ active_class_ids: selected }).eq('id', id);

            alert('Lagret deltakende klasser!');

        }


        window.nextQuestion = async function() {

            await supabase.from('game_state').update({ status: 'active', current_question: 1 }).eq('id', 1);

            alert('Neste spørsmål aktivert!');

        }


        window.lockAnswers = async function() {

            await supabase.from('game_state').update({ status: 'locked' }).eq('id', 1);

            alert('Svar låst!');

        }


        loadCompsAndClasses();

    </script>

</body>
